	@ExternImpl

class ConstructorInfo (const type, const params, const prvt, const isStatic)

	private constructor() {}

class AttributeInfo (const type, const ns, const name, const values)

    private constructor() {}

@ExternImpl
func getAttr(type) {}

@ExternImpl
func getAttr(type, prop) {}

@ExternImpl
func getAttr(type, fnc, paramsCount) {}

func getCtorAttr(type, paramsCount)
{
    return getAttr(type, type.name + ".ctor", paramsCount)
}

@ExternImpl
func getProperties(inst) {}

func getProperty(inst, name)
{
	return getProperties(inst).first(func(p) => p.name == name)
}

@ExternImpl
func setProperty(inst, name, value?) {}

@ExternImpl
func getFunctions(type) {}

func getFunction(type, name, paramsCount)
{
	const fs = getFunctions(type)
	return fs.first(func(i) => i.name == name & i.params.length() == paramsCount)
}

@ExternImpl
func getConstructors(type) {}

func getConstructor(type, paramsCount)
{
	const ctors = getConstructors(type)
	return ctors.first(func(i) => i.params.length() == paramsCount)
}
