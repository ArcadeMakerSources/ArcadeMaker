using std

namespace web:

class HttpRequest (url, method, @NullableArg headers, @NullableArg body)
{
	constructor(...);

	constructor(this.url)
	{
		method = HttpRequestMethod.get
	}

	constructor(this.url, this.method);

	constructor(this.url, this.method, this.headers?);

	@ExternImpl
	func wait(){}

	func send(onComplete?, onError?)
	{
		runAsync(func()
		{
		    var result, success = false
			try
			{
				result = this.wait()
				success = true
			}
			catch ex
			{
				if onError != null
				{
					onError(ex)
				}
			}
			if success & onComplete != null { onComplete(result) }
		})
	}
}

enum HttpRequestMethod
{
	get,
	query,
	head,
	post,
	put,
	delete,
	connect,
	options,
	trace,
	patch
}
