using std

namespace web:

class HttpRequest (url, method, headers, body)
{
	constructor(...);

	constructor(url)
	{
		this.url = url
		method = HttpRequestMethod.get
	}

	constructor(url, method)
	{
		this.url = url
		this.method = method
	}

	constructor(url, method, headers)
	{
		this.url = url
		this.method = method
		this.headers = headers
	}


	@ExternImpl
	func wait(){}

	func send(onComplete?, onError?)
	{
		runAsync(func()
		{
		    var result, success = false
			try
			{
				result = this.wait()
				success = true
			}
			catch ex
			{
				if onError != null
				{
					onError(ex)
				}
			}
			if success & onComplete != null { onComplete(result) }
		})
	}
}

enum HttpRequestMethod
{
	get,
	query,
	head,
	post,
	put,
	delete,
	connect,
	options,
	trace,
	patch
}
